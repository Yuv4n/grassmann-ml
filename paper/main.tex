\subsection{Basic Definitions}

For the definitions of smooth manifolds and smooth maps, we follow Segal \cite{segal}.

\begin{definition}\label{def:coordinate-chart}
  Let \( X \) be a topological space. A \textbf{Co-ordinate Chart} on \( X \) is the pair \( (U, \varphi)\) where \( U \subset X \) is open, and \( \varphi \) is a homeomorphism 
  \[
    \varphi: U \xlongrightarrow{\sim} \varphi(U)
  \]
  where \( \varphi(U) \) is an open subset \( \varphi(U) \subset \R^n \), for some nonnegative integer \( n \).
\end{definition}

\begin{definition}\label{def:topological-manifold}
  Let \( X \) be a topological space, and fix a nonnegative integer \( n \). \( X\) is an \textbf{n-dimensional topological manifold} if and only if \( X\) is Hausdorff and second countable, and for any point \( x \in X \), there exists a co-ordinate chart,
  \[
    \varphi: U \xlongrightarrow{\sim} \varphi(U) \subset \R^n
  \]
  where \( x \in U \), and \( \varphi(U) \) is open in \( \R^n \). 

  This is to say, that \( X \) is locally Euclidean, i.e. every point of \( X \) has an open neighbourhood homeomorphic to an open subset of \(\mathbb R^n\).

  We require that \( X \) is Hausdorff, and second countable, as to exclude certain pathological examples.
\end{definition}

\begin{definition}\label{def:transition-function}
  Let \( X \) be a topological manifold, and let \( (U, \varphi_1) \) and \( (V, \varphi_2) \) be two co-ordinate charts on \( X \). The \textbf{transition function} between these two co-ordinate charts is the function:
  \[
    \phi_{21} = \varphi_2 \circ \varphi_1^{-1} : \varphi_1(U \cap V) \xlongrightarrow{\sim} \varphi_2(U \cap V) 
  \]
  Notice, it is automatically a homeomorphism, as it is the composition of two homeomorphisms.
  
  \smallskip
  \noindent
  
  Its inverse is the transition function in the opposite direction:
  \[
    \phi_{21}^{-1} = \phi_{12} = \varphi_1 \circ \varphi_2^{-1} : \varphi_2(U \cap V) \xlongrightarrow{\sim} \varphi_1(U \cap V)
  \]
\end{definition}

\begin{definition}\label{def:atlas}
  Let \( X \) be a topological manifold. An \textbf{atlas} on \( X \) is a collection of co-ordinate charts on \( X \):
  \[
    \varphi_i : U_i \xlongrightarrow{\sim} \widetilde{U}_i \subset \R^n
  \]
  indexed by some, possibly infinite, set \( I \), such that:
  \[
  \bigcup_{i \in I}U_i = X    
  \]
  
  An atlas is a \textbf{smooth atlas} if and only if for any two co-ordinate charts in the atlas, the transition function
  \[
    \phi_{ji} = \varphi_j \circ \varphi_i^{-1} : \varphi_i(U_i \cap U_j) \xlongrightarrow{\sim} \varphi_j(U_i \cap U_j), \qquad i,j \in I
  \]
  is a smooth function, i.e. it is infinitely differentiable.
\end{definition}

\begin{definition}\label{def:compatible-atlases}
  Let \( X \) be a topological manifold, and let
  \[
    \mathcal{A} = \{(U_i, \varphi_i); i \in I\}
  \]
  be a smooth atlas on \( X \). Let \( (U, \varphi) \) be any coordinate chart on \( X \). The chart \( (U, \varphi) \) is \textbf{compatible} with the atlas \( \mathcal{A} \) if and only if the transition functions between \( (U, \varphi) \) and any chart in \( \mathcal{A} \) are diffeomorphisms, i.e. they are smooth functions with smooth inverses.

  In other words, the new chart \( (U, \varphi) \) is compatible with the atlas \( \mathcal{A} \) if and only if the union \( \mathcal{A} \cup \{(U, \varphi)\} \) is still a smooth atlas.
  
  \bigskip
  \noindent
  
  Let \( \mathcal{A}, \mathcal{B} \) be smooth atlases on \( X \). The atlases \( \mathcal{A} \) and \( \mathcal{B} \) are \textbf{compatible} if and only if every chart in \( \mathcal{A} \) is compatible with \( \mathcal{B} \), and vice versa.
\end{definition}

\begin{corollary}[{\cite[Corollary 2.20]{segal}}]\label{cor:compatibility-equivalence}
  Compatibility is an equivalence relation on smooth atlases.
\end{corollary}

\begin{definition}\label{def:smooth-manifold}
  A \textbf{smooth manifold} is a topological manifold \( X \) together with an equivalence class \( [\mathcal{A}] \) of compatible smooth atlases on \( X \). 
  
  The equivalence class \( [\mathcal{A}] \) of compatible smooth atlases is called a \textbf{smooth structure} on \( X \).

\end{definition}

\begin{definition}\label{def:smooth-map}
  Let \( X, Y \) be smooth manifolds, of dimension \( m \) and \( n \) respectively, and let 
  \[
    H: X \longrightarrow Y
  \]
  be a continuous function. Fix a point \( x \in X \). \( H \) is \textbf{smooth at \( x \)} if and only if there exists a chart \((U,\varphi)\) compatible with the smooth structure on \(X\), with \(x \in U\), and a chart \((V,\psi)\) compatible with the smooth structure on \(Y\), with \(H(U) \subset V\), such that the composition
  \[
    \psi \circ H \circ \varphi^{-1} : \varphi(U) \longrightarrow \psi(V)
  \]
  is smooth at \( \varphi(x) \). 
  
  The function \( H \) is \textbf{smooth} if and only if it is smooth at every point \( x \in X \).
\end{definition}

\begin{definition}\label{def:diffeomorphism}
  Let \(X, Y \) be smooth manifolds. A function 
  \[
    H: X \longrightarrow Y
  \]
  is a \textbf{diffeomorphism} if and only if \( H \) is a bijection, and both \( H \) and \( H^{-1} \) are smooth.

  If there exists a diffeomorphism between \( X \) and \( Y \), they are \textbf{diffeomorphic}.
\end{definition}

\begin{definition}\label{def:pseudo-chart}
  Following Segal, let \(X\) be a set. A \textbf{pseudo-chart} on \(X\) is the pair \((U,\varphi)\), where \(U \subset X\) and
  \[
    \varphi : U \longrightarrow \varphi{(U)}
  \]
  is a bijection onto an open subset
  \(\varphi{(U)} \subset \mathbb{R}^n\).

  A collection \(\mathcal{A}=\{(U_i,\varphi_i):i\in I\}\) of pseudo-charts, with the same dimension \(n\), is called a pseudo-atlas on \(X\) if
  \[
    \bigcup_{i\in I} U_i = X.
  \]
\end{definition}

\begin{proposition}[{\cite[Proposition 2.26]{segal}}]\label{prop:pseudo-atlas-topology}
  Let \( X \) be a set, and let \( \mathcal{A} = \{ (U_i,\varphi_i) : i \in I \} \) be a pseudo-atlas for \( X \). Suppose that for any pair \( (U_1, \varphi_1), (U_2, \varphi_2) \) of pseudo-charts in \( \mathcal{A} \), we have that:
  \begin{enumerate}
    \item both \( \varphi_1(U_1 \cap U_2) \subset \varphi_1(U_1) \) and \( \varphi_2(U_1 \cap U_2) \subset \varphi_2(U_2) \) are open subsets, and
    \item the transition function is continuous. 
  \end{enumerate}
  Then there exists a unique topology on \( X \) such that each pseudo-chart \( (U_i, \varphi_i) \) is a co-ordinate chart on \( X \).
\end{proposition}

\begin{corollary}[{\cite[Corollary 2.28]{segal}}]\label{cor:pseudo-atlas-smooth-structure}
  If we further assume that each transition function \( \phi_{21} \) is smooth, then \( [\mathcal{A}] \) is a smooth structure on the topological manifold \( X \), provided the resulting topology is Hausdorff and second-countable.
\end{corollary}

For embedded submanifolds and their tangent spaces, we follow the conventions of Boumal \cite{boumal}.

\begin{definition}\label{def:embedded-submanifold}
  Let \( \mathcal{E} \) be a real vector space of dimension \( d \). A non-empty subset \( \mathcal{M} \subset \mathcal{E} \) is a \textbf{smooth embedded submanifold} of \( \mathcal{E} \) of dimension \( n \), if either
  \begin{enumerate}
    \item \( n = d \) and \( \mathcal{M} \) is open in \( \mathcal{E} \), also denoted as an \textbf{open submanifold}, or
    \item \( n = d - k \) for some \( k \ge 1 \) and, for each \( x \in \mathcal{M} \), there exists a neighbourhood \( U \subset \mathcal{E} \) of \( x \) and a smooth function
    \[
      h: U \longrightarrow \mathbb{R}^k
    \]
    such that
    \begin{enumerate}
      \item If \( y \) is in \( U \), then \( h(y) = 0 \) if and only if \( y \in \mathcal{M} \); and
      \item rank Dh(x) = k, where \( Dh(x) \) is the Jacobian of \( h \) at \( x \).
    \end{enumerate}
  \end{enumerate}
  Such a function \( h \) is called a \textbf{local defining function} for \( \mathcal{M} \) at \( x \)
\end{definition}
