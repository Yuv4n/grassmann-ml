This section introduces the smooth structures and differential calculus used to construct the Stiefel and Grassmann manifolds.

\subsection{Basic Definitions}

We follow Segal \cite[Sections 2--5]{segal} for smooth structures and differential calculus.

\begin{definition}\label{def:co-ordinate-chart}
  Let \( X \) be a topological space. 
  A \textbf{co-ordinate chart} on \( X \) is the pair \( (U, \varphi)\) where \( U \subset X \) is open, and \( \varphi \) is a homeomorphism 
  \[
    \varphi: U \xlongrightarrow{\sim} \varphi(U)
  \]
  where \( \varphi(U) \) is open in \( \R^n \), for some nonnegative integer \( n \).
\end{definition}

\begin{definition}\label{def:topological-manifold}
  Let \( X \) be a topological space, and fix a nonnegative integer \( n \).
  \( X \) is an \textbf{\( n \)-dimensional topological manifold} if it is Hausdorff and second countable, and every point lies in a co-ordinate chart with image open in \( \R^n \). This last condition says that \( X \) is locally Euclidean.

  We require that \( X \) is Hausdorff, and second countable, as to exclude certain pathological examples.
\end{definition}

\begin{definition}\label{def:transition-function}
  Let \( X \) be a topological manifold, and let \( (U, \varphi_1) \) and \( (V, \varphi_2) \) be two co-ordinate charts on \( X \).
  The \textbf{transition function} between these two co-ordinate charts is the function:
  \[
    \phi_{21} = \varphi_2 \circ \varphi_1^{-1} : \varphi_1(U \cap V) \xlongrightarrow{\sim} \varphi_2(U \cap V) 
  \]
  It is a homeomorphism, with inverse \( \phi_{21}^{-1} = \phi_{12} = \varphi_1 \circ \varphi_2^{-1} \).
\end{definition}

\begin{definition}\label{def:atlas}
  Let \( X \) be a topological manifold.
  An \textbf{atlas} on \( X \) is a collection of co-ordinate charts on \( X \):
  \[
    \varphi_i : U_i \xlongrightarrow{\sim} \widetilde{U}_i \subset \R^n
  \]
  indexed by some, possibly infinite, set \( I \), such that:
  \[
  \bigcup_{i \in I}U_i = X    
  \]
  
  An atlas is a \textbf{smooth atlas} if all its transition functions are smooth, i.e. infinitely differentiable.
\end{definition}

\begin{definition}\label{def:compatible-atlases}
  Let \( X \) be a topological manifold, and let \( \mathcal{A} = \{(U_i, \varphi_i); i \in I\} \) be a smooth atlas on \( X \). 
  Let \( (U, \varphi) \) be any co-ordinate chart on \( X \).
  The chart \( (U, \varphi) \) is \textbf{compatible} with the atlas \( \mathcal{A} \) if the transition functions between \( (U, \varphi) \) and any chart in \( \mathcal{A} \) are diffeomorphisms, i.e. they are smooth functions with smooth inverses.

  In other words, the new chart \( (U, \varphi) \) is compatible with the atlas \( \mathcal{A} \) if the union \( \mathcal{A} \cup \{(U, \varphi)\} \) is still a smooth atlas.
  
  \bigskip
  \noindent
  
  Let \( \mathcal{A}, \mathcal{B} \) be smooth atlases on \( X \). The atlases \( \mathcal{A} \) and \( \mathcal{B} \) are \textbf{compatible} if every chart in \( \mathcal{A} \) is compatible with \( \mathcal{B} \), and vice versa.
\end{definition}

\begin{corollary}[{\cite[Corollary 2.20]{segal}}]\label{cor:compatibility-equivalence}
  Compatibility is an equivalence relation on smooth atlases.
\end{corollary}

\begin{definition}\label{def:smooth-manifold}
  A \textbf{smooth manifold} is a topological manifold \( X \) together with an equivalence class \( [\mathcal{A}] \) of compatible smooth atlases on \( X \). 
  
  The equivalence class \( [\mathcal{A}] \) of compatible smooth atlases is called a \textbf{smooth structure} on \( X \).

\end{definition}

\begin{definition}\label{def:smooth-map}
  Let \( X, Y \) be smooth manifolds, of dimension \( m \) and \( n \) respectively, and let \( H : X \longrightarrow Y \) be a continuous function.
  Fix a point \( x \in X \).
  \( H \) is \textbf{smooth at \( x \)} if there exists a chart \((U,\varphi)\) compatible with the smooth structure on \(X\), with \(x \in U\), and a chart \((V,\psi)\) compatible with the smooth structure on \(Y\), with \(H(U) \subset V\), such that the composition
  \[
    \psi \circ H \circ \varphi^{-1} : \varphi(U) \longrightarrow \psi(V)
  \]
  is smooth at \( \varphi(x) \). 
  
  The function \( H \) is \textbf{smooth} if it is smooth at every point \( x \in X \).
\end{definition}

\begin{definition}\label{def:diffeomorphism}
  Let \( X, Y \) be smooth manifolds. A function \( H : X \longrightarrow Y \) is a \textbf{diffeomorphism} if it is a bijection and both \( H \) and \( H^{-1} \) are smooth.
  If such a function exists, \( X \) and \( Y \) are \textbf{diffeomorphic}.
\end{definition}

\begin{definition}\label{def:pseudo-chart}
  Let \(X\) be a set. 
  A \textbf{pseudo-chart} on \(X\) is the pair \((U,\varphi)\), where \(U \subset X\) and
  \[
    \varphi : U \longrightarrow \varphi{(U)}
  \]
  is a bijection onto an open subset
  \(\varphi{(U)} \subset \mathbb{R}^n\).

  A collection \(\mathcal{A}=\{(U_i,\varphi_i):i\in I\}\) of pseudo-charts, with the same dimension \(n\), is called a pseudo-atlas on \(X\) if
  \[
    \bigcup_{i\in I} U_i = X.
  \]
\end{definition}

\begin{proposition}[{\cite[Proposition 2.26]{segal}}]\label{prop:pseudo-atlas-topology}
  Let \( X \) be a set, and let \( \mathcal{A} = \{ (U_i,\varphi_i) : i \in I \} \) be a pseudo-atlas for \( X \). Suppose that for any pair \( (U_1, \varphi_1), (U_2, \varphi_2) \) of pseudo-charts in \( \mathcal{A} \), we have that:
  \begin{enumerate}
    \item both \( \varphi_1(U_1 \cap U_2) \subset \varphi_1(U_1) \) and \( \varphi_2(U_1 \cap U_2) \subset \varphi_2(U_2) \) are open subsets, and
    \item the transition function
    \[
      \phi_{21} = \varphi_2 \circ \varphi_1^{-1} : \varphi_1(U_1 \cap U_2) \longrightarrow \varphi_2(U_1 \cap U_2)
    \]
    is continuous. 
  \end{enumerate}
  Then there exists a unique topology on \( X \) such that each pseudo-chart \( (U_i, \varphi_i) \) is a co-ordinate chart on \( X \).
\end{proposition}

\begin{corollary}[{\cite[Corollary 2.28]{segal}}]\label{cor:pseudo-atlas-smooth-structure}
  If we further assume that each transition function \( \phi_{21} \) is smooth, then \( [\mathcal{A}] \) is a smooth structure on the topological manifold \( X \), provided the resulting topology is Hausdorff and second-countable.
\end{corollary}

\begin{definition}\label{def:embedded-submanifold}
  Let \( X \) be a smooth manifold of dimension \( n \), let \( m \) be an integer satisfying \( 0 \leq m \leq n \), and let \( \mathcal{M} \subset X \).
  The subset \( \mathcal{M} \) is an \textbf{\( m \)-dimensional submanifold} of \( X \) if, for every \( x \in \mathcal{M} \), there exists a compatible co-ordinate chart
  \[
    \varphi : U \xlongrightarrow{\sim} \varphi(U) \subset \R^n
  \]
  with \( x \in U \), and an \( m \)-dimensional affine subspace \( A \subset \R^n \), such that
  \[
    \varphi(U \cap \mathcal{M}) = \varphi(U) \cap A.
  \]
  This is the convention of~\cite[Definition 3.2]{segal}. We also use the term \textbf{embedded submanifold} for a submanifold in this sense.
\end{definition}

By~\cite[Proposition 3.7]{segal}, \( \mathcal{M} \), with the subspace topology, carries a smooth structure induced from \( X \).
We always equip a submanifold with this structure. By \cite[Lemmas 4.7--4.8]{segal}, the inclusion \( \iota : \mathcal{M} \hookrightarrow X \) is smooth, and a map \( F : Y \longrightarrow \mathcal{M} \) from a smooth manifold is smooth if and only if \( \iota \circ F \) is smooth. Smooth maps on an ambient manifold therefore restrict to smooth maps on its submanifolds; in particular, a map admitting smooth local ambient extensions is smooth.

\begin{definition}\label{def:tangent-space}
  Let \( \mathcal{M} \) be a smooth manifold, and let \( x \in \mathcal{M} \). A curve through \( x \) is a smooth function \( c : I \longrightarrow \mathcal{M} \), where \( I \) is an open interval containing \( 0 \) and \( c(0) = x \).
  Two curves \( c_1, c_2 \) through \( x \) are \textbf{tangent at \( x \)} if, for any compatible co-ordinate chart \( (U, \varphi) \) containing \( x \),
  \[
    \left.\frac{d}{dt}(\varphi \circ c_1)(t)\right|_{t=0} = \left.\frac{d}{dt}(\varphi \circ c_2)(t)\right|_{t=0}.
  \]
  The compositions are taken on sufficiently small intervals around \( 0 \).
  This condition is independent of the choice of chart.

  A \textbf{tangent vector} at \( x \) is an equivalence class of curves through \( x \) under this relation.
  The \textbf{tangent space} at \( x \) is the set
  \[
    T_x\mathcal{M} = \{\text{smooth curves through } x\}/\!\sim,
  \]
  where \( c_1 \sim c_2 \) means that the curves are tangent at \( x \), following~\cite[Definitions 5.4--5.5]{segal}.
\end{definition}

By~\cite[Proposition 5.8]{segal}, \( T_x\mathcal{M} \) is a vector space of dimension \( \dim\mathcal{M} \). Its vector space structure is the one for which the chart identification
\[
  [c] \longmapsto \left.\frac{d}{dt}(\varphi \circ c)(t)\right|_{t=0}
\]
is a linear isomorphism with \( \R^{\dim\mathcal{M}} \).

For an embedded submanifold \( \mathcal{M} \) of a finite-dimensional real vector space \( \mathcal{E} \), we identify \( [c] \) with the ambient velocity \( c'(0) \), as in~\cite[Section 5.2]{segal}. Under this identification, \( T_x\mathcal{M} \) is a subspace of \( \mathcal{E} \), and
\[
  T_x\mathcal{M} = \{c'(0) : c : I \longrightarrow \mathcal{M} \text{ is smooth and } c(0) = x\}.
\]
Here, derivatives are taken in \( \mathcal{E} \).
If \( \mathcal{M} \) is open in \( \mathcal{E} \), then \( T_x\mathcal{M} = \mathcal{E} \).

\begin{definition}\label{def:derivative}
  Let \( X, Y \) be smooth manifolds, and let \( F : X \longrightarrow Y \) be smooth.
  The \textbf{derivative} of \( F \) at \( x \in X \) is the linear map
  \[
    DF(x) : T_xX \longrightarrow T_{F(x)}Y, \qquad [c] \longmapsto [F \circ c].
  \]
  This map is well-defined and linear by~\cite[Proposition 5.10]{segal}.
\end{definition}

For smooth maps \( F : X \longrightarrow Y \) and \( G : Y \longrightarrow Z \), this definition gives the chain rule
\[
  D(G \circ F)(x) = DG(F(x)) \circ DF(x).
\]

\begin{theorem}[Regular level sets {\cite[Proposition 3.16 and Lemma 5.13]{segal}}]\label{thm:regular-value}
  Let \( \mathcal{E} \) and \( \mathcal{E'} \) be finite-dimensional real vector spaces, and let \( F : \mathcal{E} \rightarrow \mathcal{E'} \) be smooth.
  A point \( c \in \mathcal{E'} \) is called a regular value of \( F \) if \( DF(x) \) is surjective for every \( x \in F^{-1}(c) \).

  If \(c\) is a regular value and \( \mathcal M = F^{-1}(c)\) is nonempty, then \(\mathcal M\) is a smooth embedded submanifold of \(\mathcal E\), with
  \[
    \dim\mathcal M=\dim\mathcal E-\dim\mathcal E', \qquad T_x\mathcal M = \ker DF(x) \quad \text{for every } x \in \mathcal{M}.
  \]
\end{theorem}

To define gradients on these manifolds, we equip their tangent spaces with inner products. For the following metric and gradient definitions, we follow \cite[Definitions 3.51--3.55, 3.58 and 3.60]{boumal}, expressing smoothness of the metric in co-ordinates.

\begin{definition}\label{def:riemannian-manifold}
  Let \( \mathcal{M} \) be a smooth manifold. A \textbf{Riemannian metric} on \( \mathcal{M} \) assigns an inner product \( \langle \cdot, \cdot \rangle_x \) to each tangent space \( T_x\mathcal{M} \), varying smoothly with \( x \).
  In co-ordinates \( (U, \varphi) \), this means that each function
  \[
    x \longmapsto {\left\langle {(D\varphi(x))}^{-1}e_i, {(D\varphi(x))}^{-1}e_j \right\rangle}_x
  \]
  is smooth on \( U \), where \( e_i, e_j \) are standard basis vectors of \( \R^{\dim\mathcal{M}} \).
  A smooth manifold equipped with a Riemannian metric is a \textbf{Riemannian manifold}.
\end{definition}

\begin{definition}\label{def:riemannian-submanifold}
  Let \(\mathcal M\) be an embedded submanifold of a finite-dimensional real vector space \(\mathcal E\) equipped with an inner product \(\langle\cdot,\cdot\rangle_{\mathcal E}\).
  The induced metric on \(\mathcal M\) is defined by
  \[
    \langle u,v\rangle_x=\langle u,v\rangle_{\mathcal E}, \qquad u,v\in T_x\mathcal M.
  \]
  This is a Riemannian metric by~\cite[Proposition 3.54]{boumal}. Equipped with this metric, \(\mathcal M\) is a \textbf{Riemannian submanifold} of \(\mathcal E\).
\end{definition}

\begin{definition}\label{def:riemannian-gradient}
  Let \( \mathcal{M} \) be a Riemannian manifold, and let \( f : \mathcal{M} \rightarrow \mathbb{R} \) be smooth.
  For each \( x \in \mathcal{M} \), the \textbf{Riemannian gradient} of \( f \) at \( x \) is the unique vector \( \grad f(x) \in T_x\mathcal{M} \) satisfying
  \[
    Df(x)[v] = \langle \grad f(x), v\rangle _x \qquad \text{for every } v \in T_x \mathcal{M}.
  \]
  Here,
  \[
    Df(x)[v] = \left.\frac{d}{dt}f(c(t))\right|_{t=0},
  \]
  where \(c\) is any smooth curve through \(x\) representing \(v\).
  This value is independent of the choice of such a curve.
\end{definition}

\begin{definition}\label{def:orthogonal-projector}
  Let \( \mathcal{M} \subset \mathcal{E} \) be an embedded submanifold, where \( \mathcal{E} \) is a finite-dimensional real inner product space, and let \( x \in \mathcal{M} \).
  The \textbf{orthogonal projector} to \( T_x\mathcal{M} \) is the linear map \( \Proj_x : \mathcal{E} \rightarrow \mathcal{E} \) characterized by the following properties:
  \begin{enumerate}
    \item Range: \( \im(\Proj_x) = T_x\mathcal{M} \).
    \item Projector: \( \Proj_x \circ \Proj_x = \Proj_x \).
    \item Orthogonal: \( \langle u - \Proj_x(u), v \rangle = 0 \quad \text{for every } v \in T_x\mathcal{M} \text{ and } u \in \mathcal{E} \).
  \end{enumerate}
\end{definition}


\begin{proposition}[{\cite[Proposition 3.61]{boumal}}]\label{prop:gradient-projection}
  Let \( \mathcal{M} \subset \mathcal{E} \) be a Riemannian submanifold, and let \( f : \mathcal{M} \rightarrow \mathbb{R} \) be smooth.
  Then the Riemannian gradient of \( f \) is given by
  \[
    \grad f(x) = \Proj_x\bigl(\nabla \bar{f}(x)\bigr), \qquad x \in \mathcal{M},
  \]
  where \( \bar{f} \) is any smooth extension of \( f \) to an open neighbourhood of \( \mathcal{M} \) in \( \mathcal{E} \), and \( \nabla \bar{f}(x) \) is its Euclidean gradient.

  The Euclidean gradient is characterized by
  \[
    D\bar{f}(x)[v] = \langle \nabla \bar{f}(x), v \rangle \qquad \text{for every } v \in \mathcal{E}.
  \]
\end{proposition}

A gradient gives a tangent direction at a point.
Retractions allow us to use this direction to move on the manifold.

\begin{definition}\label{def:retraction}
  Let \( \mathcal{M} \) be a smooth manifold.
  The \textbf{tangent bundle} of \( \mathcal{M} \) is the set
  \[
    T\mathcal{M} = \{(x, v) : x \in \mathcal{M},\ v \in T_x\mathcal{M}\}.
  \]
  It carries a natural smooth structure by~\cite[Proposition E.1]{segal}. Following \cite[Definition 3.47]{boumal}, a \textbf{retraction} on \( \mathcal{M} \) is a smooth map
  \[
    R : T\mathcal{M} \rightarrow \mathcal{M} : (x, v) \mapsto R_x(v)
  \]
  such that, for every \( x \in \mathcal{M} \) and \( v \in T_x\mathcal{M} \), the curve \( c(t) = R_x(tv) \) satisfies \( c(0) = x \) and represents \( v \). 
  
  For an embedded submanifold \( \mathcal{M} \subset \mathcal{E} \), we regard \( R \) as a map into \( \mathcal{E} \).
  It is smooth if and only if it extends to a smooth map on an open neighbourhood of \( T\mathcal{M} \) in \( \mathcal{E} \times \mathcal{E} \).

\end{definition}


We will also need to determine when the derivative of a smooth map is surjective.


\begin{definition}\label{def:submersion}
  Let \( X, Y \) be smooth manifolds.
  A smooth function \( F : X \longrightarrow Y \) is a \textbf{submersion} if
  \[
    \rank DF(x) = \dim Y \qquad \text{for every } x \in X.
  \]
  Equivalently, \( DF(x) : T_x X \longrightarrow T_{F(x)}Y \) is surjective at every point, as in~\cite[Definition 4.16 and Section 5.1]{segal}.
\end{definition}

\subsection{Examples}
\begin{example}[Sphere]\label{ex:sphere}\label{ex:sphere-tangent-space}
  Following \cite[Examples 3.19 and 5.14]{segal}, consider the smooth function
  \[
    h : \R^{n+1} \longrightarrow \R, \qquad (x_0, \dots, x_n) \mapsto x_0^2 + \cdots + x_n^2.
  \]
  The unit sphere is \( S^n = h^{-1}(1) \). The derivative of \( h \) at \( x = (x_0, \dots, x_n) \) is the 1-by-\( (n+1) \) matrix
  \[
    Dh(x) = (2x_0, \dots, 2x_n) : \R^{n+1} \longrightarrow \R,
  \]
  which is surjective if at least one of the \( x_i \) is nonzero. Every point of \( S^n \) is nonzero, so \( 1 \) is a regular value of \( h \). By \cref{thm:regular-value}, \( S^n \) is an \( n \)-dimensional embedded submanifold of \( \R^{n+1} \).

  The same theorem gives its tangent space at \( x \in S^n \):
  \[
    T_xS^n = \ker Dh(x) = \{v \in \R^{n+1} : x \cdot v = 0\} = x^\perp.
  \]
\end{example}

\begin{example}[Real projective space]\label{ex:real-projective-space}
  Following \cite[Example 2.30]{segal}, let \( \mathbb{RP}^n \) denote the set of lines through the origin in \( \R^{n+1} \). For a nonzero vector \( x = (x_0, \dots, x_n) \), we denote the corresponding line by \( [x] = x_0 : \cdots : x_n \).
  For each \( i \in \{0, \dots, n\} \), set
  \[
    U_i = \{x_0 : \cdots : x_n : x_i \ne 0\}.
  \]
  These sets cover \( \mathbb{RP}^n \), since every nonzero vector has a nonzero component. Define
  \[
    f_i : U_i \longrightarrow \R^n, \qquad
    (x_0 : \cdots : x_n) \mapsto
    \left(\frac{x_0}{x_i}, \dots, \frac{x_{i-1}}{x_i}, \frac{x_{i+1}}{x_i}, \dots, \frac{x_n}{x_i}\right).
  \]
  The ratios are unchanged when \( x \) is multiplied by a nonzero scalar, so \( f_i \) is well-defined. It is a bijection, with inverse
  \[
    f_i^{-1} : (\tilde{x}_1, \dots, \tilde{x}_n) \mapsto
    \tilde{x}_1 : \cdots : \tilde{x}_i : 1 : \tilde{x}_{i+1} : \cdots : \tilde{x}_n.
  \]
  Write \( z_i(\tilde{x}) \in \R^{n+1} \) for the vector obtained by inserting \( 1 \) into coordinate \( i \), with coordinates indexed from \( 0 \) to \( n \). Thus \( f_i^{-1}(\tilde{x}) = [z_i(\tilde{x})] \).

  For any \( i,j \), the overlap image is
  \[
    f_i(U_i \cap U_j) = \{\tilde{x} \in \R^n : (z_i(\tilde{x}))_j \ne 0\},
  \]
  which is open. On this set, the transition function is
  \[
    (f_j \circ f_i^{-1})(\tilde{x})
    = \left(\frac{(z_i(\tilde{x}))_r}{(z_i(\tilde{x}))_j}\right)_{r\ne j},
  \]
  with components ordered by \( r \in \{0,\dots,n\} \). Each transition is smooth because its denominator is nonzero on the overlap. By \cref{prop:pseudo-atlas-topology}, these pseudo-charts determine a unique topology in which they are co-ordinate charts.

  To check Hausdorffness, consider
  \[
    P : \mathbb{RP}^n \longrightarrow \Sym(n+1), \qquad
    [x] \longmapsto \frac{xx^\top}{\|x\|^2},
  \]
  where \( \Sym(n+1) \) is the vector space of real symmetric \( (n+1) \)-by-\( (n+1) \) matrices. This map is well-defined because scaling \( x \) does not change the quotient. It is injective because the image of the matrix \( P([x]) \) is the line \( [x] \). In the chart \( U_i \),
  \[
    (P \circ f_i^{-1})(\tilde{x})
    = \frac{z_i(\tilde{x})z_i(\tilde{x})^\top}{1+\|\tilde{x}\|^2},
  \]
  which is continuous. Thus \( P \) is a continuous injection into the Hausdorff space \( \Sym(n+1) \), so \( \mathbb{RP}^n \) is Hausdorff.

  Each \( U_i \) has a countable basis transported from \( \R^n \). Since the \( U_i \) form a finite open cover, the union of these bases is a countable basis for \( \mathbb{RP}^n \). By \cref{cor:pseudo-atlas-smooth-structure}, the charts therefore give \( \mathbb{RP}^n \) the structure of a smooth manifold of dimension \( n \).

  More generally, the set of \( k \)-dimensional subspaces of \( \R^{n+1} \) is the Grassmannian \( \mathrm{Gr}(k,n+1) \). Real projective space \( \mathbb{RP}^n \) is the case \( k=1 \).
\end{example}
